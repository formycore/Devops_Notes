step 1: MYSQL pod
    kubectl create configmap db-cm --from-literal=MYSQL_DATABASE=mydb -n todo
    kubectl create secret generic db-secret --from-literal=MYSQL_ROOT_PASSWORD=rootpassword -n todo
    kubectl run mysqlpod --image=mysql:5.7 --dry-run=client -o yaml > mysqlpod.yaml
-------------------
cat mysqlpod.yaml 
apiVersion: v1
kind: Pod
metadata:
  labels:
    run: mysqlpod
  name: mysqlpod
  namespace: todo
spec:
  containers:
  - image: mysql:5.7
    name: mysqlpod
    ports:
      - containerPort: 3306
    envFrom:
      - configMapRef:
          name: db-cm
      - secretMapRef:
          name: db-secret
    resources: {}
  dnsPolicy: ClusterFirst
  restartPolicy: Always
status: {}
---------------------
kubectl expose pod mysqlpod \
--port=3306 \
--targetPort=3306 \
--name=mysql-svc \
-n todo
